cmake_minimum_required(VERSION 3.20)
project(reel_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/Release)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/Release)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/Release)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/Release)
execute_process(COMMAND node -p "require('node-addon-api').include" WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE NAPI_INCLUDE OUTPUT_STRIP_TRAILING_WHITESPACE)
string(REPLACE "\"" "" NAPI_INCLUDE ${NAPI_INCLUDE})
add_library(reel_engine SHARED native/addon.cpp native/engine/Engine.cpp ${CMAKE_JS_SRC})
set_target_properties(reel_engine PROPERTIES PREFIX "" SUFFIX ".node")
target_include_directories(reel_engine PRIVATE ${CMAKE_JS_INC} ${NAPI_INCLUDE} native/engine/include)
target_compile_definitions(reel_engine PRIVATE NAPI_VERSION=8 NAPI_CPP_EXCEPTIONS BOOST_ASIO_ENABLE_CANCELIO BOOST_ASIO_HAS_STD_CHRONO BOOST_ASIO_NO_DEPRECATED BOOST_EXCEPTION_DISABLE TORRENT_SSL_PEERS TORRENT_USE_LIBCRYPTO TORRENT_USE_OPENSSL _FILE_OFFSET_BITS=64)
if(APPLE)
  target_include_directories(reel_engine PRIVATE Vendor/include)
  target_link_libraries(reel_engine PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/Vendor/libtorrent-rasterbar.a ${CMAKE_CURRENT_SOURCE_DIR}/Vendor/libssl.a ${CMAKE_CURRENT_SOURCE_DIR}/Vendor/libcrypto.a "-framework Security" "-framework SystemConfiguration" "-framework CoreFoundation")
  target_link_options(reel_engine PRIVATE "-undefined" "dynamic_lookup")
else()
  find_package(LibtorrentRasterbar CONFIG REQUIRED)
  target_link_libraries(reel_engine PRIVATE LibtorrentRasterbar::torrent-rasterbar ${CMAKE_JS_LIB})
endif()
